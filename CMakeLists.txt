cmake_minimum_required(VERSION 3.20)
project(armor_swap LANGUAGES C CXX)

if(NOT OpenRACMod_DIR)
  if(DEFINED OPENRAC_SDK)
    set(OpenRACMod_DIR "${OPENRAC_SDK}/cmake")
  else()
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/sdk.lock" lock)
    string(JSON sdk_version GET "${lock}" version)
    set(OpenRACMod_DIR "${CMAKE_CURRENT_SOURCE_DIR}/.sdk/openrac-mod-sdk-${sdk_version}/cmake")
  endif()
endif()
find_package(OpenRACMod 1.34 REQUIRED)
openrac_add_mod(armor_swap MANIFEST openrac-mod.json SOURCES src/mod.cpp)
target_include_directories(armor_swap PRIVATE src)

include(CTest)
if(BUILD_TESTING)
  add_executable(selection_test tests/selection_test.cpp)
  target_compile_features(selection_test PRIVATE cxx_std_17)
  target_include_directories(selection_test PRIVATE src)
  add_test(NAME selection COMMAND selection_test)

  add_executable(native_vendor_test tests/native_vendor_test.cpp)
  target_compile_features(native_vendor_test PRIVATE cxx_std_17)
  target_include_directories(native_vendor_test PRIVATE src)
  add_test(NAME native_vendor COMMAND native_vendor_test)
endif()
